# 微机原理



```c


#pragma vector = PORT1_VECTOR
__interrupt void Port_1(void)
{
	if(P1IFG & BIT2){
		P8OUT ^= BIT1;
		P1IFG &= ~BIT2;
	}
	else{
		P3OUT ^= BIT7;
		P1IFG &= ~BIT3;
	}
	//或者 P1IFG &= ~（BIT2+BIT3); //清空中断标志位
	P1IE |= BIT2 +BIT3; //中断允许
}
```
